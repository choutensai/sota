\nomenclature[T]{MRS}{Multi-Robot System – Hệ thống đa robot.}
\nomenclature[T]{COM}{Center of Mass – Trọng tâm của vật.}
\nomenclature[T]{LS}{Least Squares – Phương pháp bình phương tối thiểu.}
\nomenclature[T]{QP}{Quadratic Programming – Quy hoạch toàn phương.}

\nomenclature[T]{PSO}{Particle Swarm Optimization – Tối ưu bầy đàn hạt.}
\nomenclature[T]{DPSO}{Distributed PSO – Tối ưu bầy đàn hạt phân tán.}
\nomenclature[T]{ALM}{Augmented Lagrangian Method – Phương pháp Lagrangian tăng cường.}
\nomenclature[T]{ALPSO}{Augmented Lagrangian PSO – Tối ưu bầy đàn hạt phân tán kết hợp Lagrangian tăng cường.}
\nomenclature[T]{DALPSO}{Distributed ALPSO – Tối ưu bầy đàn hạt kết hợp Lagrangian tăng cường.}

\nomenclature[T]{GWS}{Grasp Wrench Space – Không gian wrench khả thi của cấu hình tiếp xúc.}

% =========================
% Ký hiệu (Group K)
% =========================
\nomenclature[K]{$n$}{Số lượng robot tham gia nhiệm vụ.}

\nomenclature[K]{$\mathbf{u}$}{Vector cấu hình (vị trí tiếp xúc) của $n$ robot trên biên vật.}
\nomenclature[K]{$\mathcal{U}$}{Không gian vị trí của robot.}

\nomenclature[K]{$f_i$}{Độ lớn lực đẩy của robot $i$ (đẩy vuông góc cạnh tiếp xúc).}
\nomenclature[K]{$\mathbf{f}_r$}{Vector lực đẩy của toàn bộ robot.}
\nomenclature[K]{$f_{\max}$}{Giới hạn lực đẩy cực đại.}

\nomenclature[K]{$\mathbf{w}$}{Wrench tại khối tâm.}
\nomenclature[K]{$\mathbf{w}_d$}{Wrench mong muốn.}
\nomenclature[K]{$\mathbf{w}_e$}{Wrench tổng hợp tại khối tâm do các robot sinh ra.}

\nomenclature[K]{$\mathbf{w}_i$}{Wrench do tiếp xúc/robot $i$ tác dụng lên vật.}
\nomenclature[K]{$\mathbf{r}_i$}{Cánh tay đòn từ COM đến tiếp xúc $i$.}
\nomenclature[K]{$\lambda$}{Hệ số chuẩn hoá chiều mô men.}

\nomenclature[K]{$\mathbf{G}$}{Ma trận ánh xạ lực tiếp xúc sang wrench tại khối tâm.}

\nomenclature[K]{$\theta(t)$}{Góc quay của vật theo thời gian.}
\nomenclature[K]{$\dot{\theta}$}{Vận tốc góc của vật.}
\nomenclature[K]{$\ddot{\theta}$}{Gia tốc góc của vật.}
\nomenclature[K]{$I$}{Mô men quán tính quay của vật quanh COM.}
\nomenclature[K]{$\tau_{fr}$}{Mô men ma sát quay (Coulomb/giới hạn ma sát quay).}
\nomenclature[K]{$\Delta r_i$}{Độ thay đổi cánh tay đòn giữa các lần đẩy thử.}

\nomenclature[K]{$J(\mathbf{u})$}{Hàm mục tiêu đánh giá cấu hình.}
\nomenclature[K]{$J_t(\mathbf{u})$}{Độ ổn định tịnh tiến của cấu hình.}
\nomenclature[K]{$J_r(\mathbf{u})$}{Độ ổn định xoay của cấu hình.}

\nomenclature[K]{$d_{\min}$}{Khoảng cách an toàn tối thiểu giữa các robot/tiếp xúc.}


\nomenclature[K]{$L(\mathbf{u},\boldsymbol{\lambda},\rho)$}{Hàm Augmented Lagrangian cho bài toán tối ưu ràng buộc.}
\nomenclature[K]{$\rho$}{Hệ số tăng cường/phạt bậc hai trong ALM/ALPSO.}

\nomenclature[K]{$\mu_s$}{Hệ số ma sát tĩnh (Coulomb) tại tiếp xúc.}
\nomenclature[K]{$\mu_k$}{Hệ số ma sát động (Coulomb) tại tiếp xúc.}
\nomenclature[K]{$\gamma$}{Hệ số phản ánh khả năng truyền mô men của vật liệu tiếp xúc.}
