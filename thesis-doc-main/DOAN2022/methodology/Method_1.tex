\section{Methodology}
\subsection{Formulation}
The system consists of $N$ robots maninupulating an object with polygonal shape. Each robot can exert a normal force $f_i$ within a limited range : $0 < f_i < f_{max}$ at its contact point $C_i$.The forces all
robots exert on the object are cast into the vector $f_r= [f_1, \ldots, f_N]$  A formation $p = [p_{C_1}, \ldots, p_{C_N}]$ 

\begin{equation}
M
=
\sum_{i=1}^{n}
\mathbf{L}_{i} \times \mathbf{F}_{i}
+ M_f
\end{equation}

\begin{equation}
M
=
\sum_{i=1}^{n}
\mathbf{p}_{i} \times \mathbf{F}_{i}
-
\mathbf{p}_{C} \times
\sum_{i=1}^{n}
\mathbf{F}_{i}
+
M_f
\end{equation}

\begin{equation}
M
=
\sum_{i=1}^{n}
\mathbf{p}_{i} \times \mathbf{F}_{i}
+
M_f
\end{equation}

\begin{equation}
G(u)
=
\begin{bmatrix}
\mathbf{n}(u_1) & \cdots & \mathbf{n}(u_N) \\
\ell(u_1) & \cdots & \ell(u_N)
\end{bmatrix}
\in \mathbb{R}^{3 \times N}
\end{equation}

\begin{equation}
\mathbf{w}
=
G(\mathbf{u})\mathbf{f}
=
\begin{bmatrix}
\mathbf{n}(u_1) & \cdots & \mathbf{n}(u_N) \\
\ell(u_1) & \cdots & \ell(u_N)
\end{bmatrix}
\begin{bmatrix}
f_1\\
\vdots\\
f_N
\end{bmatrix}
=
\begin{bmatrix}
F_x\\
F_y\\
M
\end{bmatrix}
\in \mathbb{R}^{3}.
\end{equation}

\begin{equation}
\begin{aligned}
\min_{\mathbf{u} \in \mathbb{R}_{\geq 0}^{N}}
\quad & J(\mathbf{u}) = -J_r(\mathbf{u}) \\
\text{s.t.}\quad
& \exists\,\tilde{\mathbf{f}} \in \mathcal{F}
\quad \text{s.t.}\quad
G(\mathbf{u})\tilde{\mathbf{f}} = \mathbf{w}_d, \\
& \mathrm{frD}(u_i,u_j) \geq d_{\min},
\qquad \forall i \neq j .
\end{aligned}
\end{equation}
