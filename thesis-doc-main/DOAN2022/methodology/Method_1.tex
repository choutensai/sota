\section{Methodology}

\begin{equation}
\ddot{\mathbf{p}}_C
=
\frac{1}{m}
\left(
\sum_{i=1}^{n}\mathbf{F}_i+\mathbf{F}_f
\right)
\end{equation}

\begin{equation} 
\ddot{\Theta} 
= 
\frac{1}{I}
\left[
\sum_{i=1}^{n} 
\left(\mathbf{p}_{C_i}-\mathbf{p}_{C}\right) 
\times \mathbf{F}_{i}
+ M_f
\right]
\end{equation}

\begin{equation}
\ddot{\Theta}
=
\frac{1}{I}
\sum_{i=1}^{N}
\mathbf{p}_{C_i}\times\mathbf{F}_i
-
\frac{1}{I}
\mathbf{p}_C\times
\sum_{i=1}^{N}\mathbf{F}_i
+
\frac{1}{I}M_f
\end{equation}
